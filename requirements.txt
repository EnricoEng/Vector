# Geração da representação visual do grafo de chamadas.
#
# O pacote apenas escreve o arquivo DOT e invoca o programa externo
# "dot". Esse programa é instalado separadamente, a partir de
# https://graphviz.org/download/. Sem ele, a PoC continua funcionando
# e grava somente o arquivo DOT.
graphviz==0.21

# Análise sintática de código C.
#
# As versões abaixo estão fixadas porque o tree-sitter valida a versão
# da ABI da gramática: combinações mais recentes de tree-sitter-c são
# recusadas por versões mais antigas de tree-sitter, com o erro
# "Incompatible Language version".
tree-sitter==0.23.2
tree-sitter-c==0.23.4

# Tema moderno para a interface gráfica.
#
# Dependência opcional. Sem ela, a janela usa o tema "clam" do próprio
# ttk e todas as funcionalidades permanecem idênticas.
#
# Exige Tk 8.6 ou superior: o ttkbootstrap desenha seus widgets a partir
# de imagens PNG, e em versões anteriores a criação da janela falha. A
# interface detecta essa situação e volta ao tema padrão sozinha.
ttkbootstrap==1.18.2
